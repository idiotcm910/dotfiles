# tmux-grid Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Một phím `prefix g` gộp mọi window của session tmux hiện tại thành lưới pane trong một window, bấm lại để tách mỗi pane về đúng window cũ.

**Architecture:** Một script bash duy nhất `bin/tmux-grid` chứa toàn bộ logic, được `tmux.conf` gọi qua `run-shell`. Trạng thái lưu trong chính tmux bằng user option (`@grid` trên window, `@grid_win_*` trên từng pane) — không có file tạm. Kiểm thử bằng script bash chạy trên một tmux server riêng biệt qua socket `-L gridtest`.

**Tech Stack:** bash, tmux 3.4, fzf-tmux (đã cài sẵn tại `/usr/bin/fzf-tmux`).

**Spec:** `docs/specs/2026-07-21-tmux-grid-merge-windows-design.md`

## Global Constraints

- **TUYỆT ĐỐI không chạy `tmux kill-server` hay `tmux kill-session` mà thiếu cờ `-L gridtest`.** Server tmux mặc định là nơi người dùng đang làm việc thật; giết nó là mất toàn bộ session đang chạy. Mọi lệnh tmux trong file test phải đi qua hàm bọc `T()`.
- Script chạy được trên tmux 3.4, bash 5.x. Không thêm phụ thuộc mới ngoài `fzf-tmux` đã có.
- Toàn bộ chữ hiển thị cho người dùng (thông báo, prompt, doc) viết bằng tiếng Việt có dấu, khớp giọng văn các script `wt-*` sẵn có.
- Script mới đặt tại `bin/`, đã nằm trong `PATH` nhờ `setup-workspace.sh` — không thêm bước cài đặt.
- Ba sự thật về tmux đã kiểm chứng thực nghiệm trên máy này, code phải tôn trọng:
  1. `renumber-windows on` đang bật → `window_index` thay đổi ngay khi một window bị rỗng và biến mất. **Phải thu thập toàn bộ metadata trước khi join pane đầu tiên.**
  2. Pane user option (`set-option -p`) sống sót qua `join-pane` và `break-pane`.
  3. `run-shell` **không** set `TMUX_PANE` và **không** truyền target `-t` vào script con. Script luôn thao tác trên window đang active của session — đúng với hành vi mong muốn khi bind phím.

```bash
#!/usr/bin/env bash
# tests/tmux-grid.test.sh — kiểm bin/tmux-grid trên tmux server RIÊNG (socket gridtest).
# CẢNH BÁO: mọi lệnh tmux phải đi qua T(); gọi tmux trần sẽ đụng session thật của người dùng.
set -uo pipefail

SOCK=gridtest
REPO="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
GRID="$REPO/bin/tmux-grid"
fail=0

T() { tmux -L "$SOCK" "$@"; }
ok() { printf "  \033[32m✓\033[0m %s\n" "$1"; }
no() { printf "  \033[31m✗\033[0m %s\n      mong đợi: %s\n      thực tế : %s\n" "$1" "$2" "$3"; fail=1; }
eq() { [ "$2" = "$3" ] && ok "$1" || no "$1" "$2" "$3"; }

# mkwins <session> <tên window>... — dựng session sạch với các window cho trước
mkwins() {
  local s="$1"; shift
  T kill-session -t "$s" 2>/dev/null
  T new-session -d -s "$s" -n "$1"; shift
  local w; for w in "$@"; do T new-window -t "$s": -n "$w"; done
}

# grid <session> <window active> [layout] — chạy tmux-grid như tmux sẽ chạy (run-shell)
grid() {
  local s="$1" w="$2" layout="${3:-}"
  T select-window -t "$s:$w"
  T run-shell "$GRID $layout"
}

cleanup() { T kill-server 2>/dev/null; }
trap cleanup EXIT
cleanup

echo "── Gộp ──"
mkwins t1 alpha beta gamma
grid t1 alpha tiled
eq "session còn 1 window"        "1" "$(T list-windows -t t1 -F x | wc -l)"
eq "window đó có 3 pane"         "3" "$(T list-panes -t t1 -F x | wc -l)"
eq "window được đánh cờ @grid"   "on" "$(T display-message -p -t t1 '#{@grid}')"
eq "pane nhớ tên window nguồn"   "beta gamma" \
   "$(T list-panes -t t1 -F '#{@grid_win_name}' | grep -v '^$' | tr '\n' ' ' | sed 's/ *$//')"

echo ""
[ "$fail" -eq 0 ] && echo "TẤT CẢ ĐỀU ĐẠT" || echo "CÓ TEST HỎNG"
exit "$fail"
```

```bash
#!/usr/bin/env bash
# tmux-grid — gộp mọi window của session hiện tại thành lưới pane trong một window;
# chạy lại để tách mỗi pane về đúng window cũ.
#
# Dùng: prefix g (bind trong tmux.conf). Truyền tên layout làm tham số để chạy
# không tương tác, bỏ qua popup — dành cho test.
set -euo pipefail

[ -n "${TMUX:-}" ] || { echo "tmux-grid: phải chạy bên trong tmux" >&2; exit 1; }

TAB=$'\t'

# Window đang active của session. run-shell không truyền target nên không dùng -t.
cur_win() { tmux display-message -p '#{window_id}'; }

merge() {
  local dst="$1" layout="$2"
  local rows=() r p wid wix wnm last

  # Thu thập TRƯỚC khi join: renumber-windows on làm window_index đổi ngay khi
  # một window bị hút rỗng, đọc sau sẽ ra số sai.
  mapfile -t rows < <(
    tmux list-panes -s -F "#{pane_id}${TAB}#{window_id}${TAB}#{window_index}${TAB}#{window_name}" \
      | awk -F'\t' -v d="$dst" '$2 != d'
  )
  [ "${#rows[@]}" -gt 0 ] || return 0

  for r in "${rows[@]}"; do
    IFS="$TAB" read -r p wid wix wnm <<<"$r"
    tmux set-option -p -t "$p" @grid_win_id   "$wid"
    tmux set-option -p -t "$p" @grid_win_name "$wnm"
    tmux set-option -p -t "$p" @grid_win_idx  "$wix"
    # Join vào pane cuối cùng để pane xếp đúng thứ tự window gốc.
    last="$(tmux list-panes -t "$dst" -F '#{pane_id}' | tail -1)"
    tmux join-pane -d -s "$p" -t "$last"
  done

  tmux select-layout -t "$dst" "$layout"
  tmux set-option -w -t "$dst" @grid on
  tmux set-option -w -t "$dst" @grid_layout "$layout"
}

main() {
  local dst; dst="$(cur_win)"
  merge "$dst" "${1:-tiled}"
}

main "$@"
```

```bash
split_back() {
  local dst="$1"
  local rows=() r p gid gix gnm
  declare -A newwin=()

  # Chỉ những pane có @grid_win_id mới là pane bị hút vào; pane do người dùng
  # tự split sau khi gộp không có cờ nên được giữ nguyên tại window hiện tại.
  mapfile -t rows < <(
    tmux list-panes -t "$dst" -F "#{pane_id}${TAB}#{@grid_win_id}${TAB}#{@grid_win_idx}${TAB}#{@grid_win_name}" \
      | awk -F'\t' '$2 != ""' \
      | sort -t"$TAB" -k3,3n
  )

  for r in "${rows[@]}"; do
    IFS="$TAB" read -r p gid gix gnm <<<"$r"
    if [ -z "${newwin[$gid]:-}" ]; then
      tmux break-pane -d -s "$p" -n "$gnm"
      newwin[$gid]="$(tmux display-message -p -t "$p" '#{window_id}')"
    else
      tmux join-pane -d -s "$p" -t "${newwin[$gid]}"
    fi
    tmux set-option -pu -t "$p" @grid_win_id
    tmux set-option -pu -t "$p" @grid_win_name
    tmux set-option -pu -t "$p" @grid_win_idx
  done

  tmux set-option -wu -t "$dst" @grid
  tmux set-option -wu -t "$dst" @grid_layout
}
```

Thay hàm `main` bằng:

```bash
main() {
  local dst; dst="$(cur_win)"
  if [ "$(tmux display-message -p -t "$dst" '#{@grid}')" = "on" ]; then
    split_back "$dst"
  else
    merge "$dst" "${1:-tiled}"
  fi
}
```

```bash
# Chọn layout. In ra stdout tên layout, hoặc chuỗi rỗng nếu người dùng thoát fzf.
pick_layout() {
  printf '%s\n' "$LAYOUTS" \
    | fzf-tmux -p 42%,40% --no-sort --prompt '  layout  ' --border-label " gộp $1 pane " \
    || true
}

# Hỏi xác nhận khi lưới sẽ quá chật. Trả về 0 nếu đồng ý.
confirm_many() {
  local pick
  pick="$(printf 'Huỷ\nGộp %s pane\n' "$1" | fzf-tmux -p 42%,30% --no-sort --prompt '  lưới sẽ chật  ' || true)"
  [ "$pick" = "Gộp $1 pane" ]
}
```
